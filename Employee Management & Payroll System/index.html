1. ES6 Class & Constructor (new Employee(...))

Creates structured employee entities with properties & salary computation methods:

class Employee {
  constructor(id, name, department, baseSalary, rating) {
    this.id = id;
    this.name = name;
    this.department = department;
    this.baseSalary = Number(baseSalary);
    this.rating = Number(rating);
  }

  // Bonus rate based on performance rating (1 to 5)
  getBonusRate() {
    const rates = { 5: 0.20, 4: 0.15, 3: 0.10, 2: 0.05, 1: 0.00 };
    return rates[this.rating] || 0;
  }
}

3. Arrow Functions (Net Salary & Tax Deductions)

Concise functional formulas calculating bonuses, tax deductions, and final net earnings:

// Arrow function to calculate bonus amount
const calculateBonus = (baseSalary, rating) => {
  const bonusRates = { 5: 0.20, 4: 0.15, 3: 0.10, 2: 0.05, 1: 0.00 };
  return baseSalary * (bonusRates[rating] || 0);
};

// Arrow function to calculate tax deduction
const calculateTax = (grossSalary) => grossSalary * DEFAULT_TAX_RATE;

// Arrow function to compute complete net salary breakdown
const calculateNetSalary = (emp) => {
  const bonus = calculateBonus(emp.baseSalary, emp.rating);
  const gross = emp.baseSalary + bonus;
  const tax = calculateTax(gross);
  return gross - tax;
};

7. Aggregating Statistics with reduce()

Calculates Total Payroll Cost, Average Net Salary, and Top Earner in a single functional pass:

const calculateCompanyStats = (employees) => {
  if (employees.length === 0) return { totalPayroll: 0, avgSalary: 0, highestPaid: null };

  // 1. Total Payroll Cost (Sum of Net Salaries) via reduce()
  const totalPayroll = employees.reduce((accumulator, emp) => {
    return accumulator + calculateNetSalary(emp);
  }, 0);

  // 2. Average Net Salary
  const avgSalary = totalPayroll / employees.length;

  // 3. Highest Paid Employee via reduce()
  const highestPaid = employees.reduce((highest, current) => {
    const highestNet = calculateNetSalary(highest);
    const currentNet = calculateNetSalary(current);
    return currentNet > highestNet ? current : highest;
  }, employees[0]);

  return { totalPayroll, avgSalary, highestPaid };
};
